# ===============================
# Job Fitness AI - Data Cleaning
# ===============================

import pandas as pd
import numpy as np

# -------------------------------
# 1. Load Dataset
# -------------------------------
df = pd.read_csv("career_dataset.csv")

print("Original Dataset Shape:", df.shape)

# -------------------------------
# 2. Display Basic Information
# -------------------------------
print("\nDataset Information")
print(df.info())

print("\nFirst 5 Rows")
print(df.head())

# -------------------------------
# 3. Check Missing Values
# -------------------------------
print("\nMissing Values")
print(df.isnull().sum())

# -------------------------------
# 4. Remove Duplicate Records
# -------------------------------
duplicates = df.duplicated().sum()
print("\nDuplicate Rows:", duplicates)

df.drop_duplicates(inplace=True)

# -------------------------------
# 5. Remove Extra Spaces
# -------------------------------
for col in df.select_dtypes(include='object').columns:
    df[col] = df[col].str.strip()

# -------------------------------
# 6. Standardize Text Columns
# -------------------------------
text_columns = [
    "Gender",
    "Education",
    "Degree",
    "Experience_Level",
    "Internship_Experience",
    "Problem_Solving",
    "English_Communication",
    "Interest",
    "Preferred_Work_Mode",
    "Recommended_Job"
]

for col in text_columns:
    if col in df.columns:
        df[col] = df[col].str.title()

# -------------------------------
# 7. Fill Missing Values
# -------------------------------

# Numerical Columns
numerical_cols = df.select_dtypes(include=np.number).columns

for col in numerical_cols:
    df[col].fillna(df[col].median(), inplace=True)

# Categorical Columns
categorical_cols = df.select_dtypes(include="object").columns

for col in categorical_cols:
    df[col].fillna("Unknown", inplace=True)

# -------------------------------
# 8. Remove Unnecessary Columns
# -------------------------------
if "Student_ID" in df.columns:
    df = df.drop("Student_ID", axis=1)

# -------------------------------
# 9. Convert Numeric Columns
# -------------------------------
numeric_columns = [
    "Age",
    "CGPA",
    "Years_of_Experience",
    "Projects_Count",
    "Resume_Score",
    "Coding_Test_Score",
    "Interview_Score"
]

for col in numeric_columns:
    if col in df.columns:
        df[col] = pd.to_numeric(df[col], errors="coerce")

# -------------------------------
# 10. Handle Outliers
# -------------------------------

if "Age" in df.columns:
    df = df[(df["Age"] >= 18) & (df["Age"] <= 35)]

if "CGPA" in df.columns:
    df = df[(df["CGPA"] >= 5.5) & (df["CGPA"] <= 10)]

if "Projects_Count" in df.columns:
    df = df[(df["Projects_Count"] >= 0) & (df["Projects_Count"] <= 15)]

# -------------------------------
# 11. Check Missing Values Again
# -------------------------------
print("\nMissing Values After Cleaning")
print(df.isnull().sum())

# -------------------------------
# 12. Final Dataset Information
# -------------------------------
print("\nCleaned Dataset Shape:", df.shape)

print("\nData Types")
print(df.dtypes)

# -------------------------------
# 13. Save Clean Dataset
# -------------------------------
df.to_csv("career_dataset_cleaned.csv", index=False)

print("\nData Cleaning Completed Successfully!")
print("Cleaned dataset saved as 'career_dataset_cleaned.csv'")
